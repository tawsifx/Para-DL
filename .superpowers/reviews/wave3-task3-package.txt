@pytest.mark.slow
def test_watchdog_recycle_workflow_end_to_end(monkeypatch: pytest.MonkeyPatch) -> None:
    """Compose fake time + fake aria2 RPC into one full watchdog recycle workflow.

    Drives a real ``WatchdogController`` across a multi-second window using
    ``FakeClock`` (no real sleeps). A ``FakeAria2Server`` supplies per-tick
    progress/status and receives the force-pause/unpause recycle commands. A
    ``watchdog_action`` event is emitted through the real ``UI.emit_event``
    surface (captured by a spy). The composed state sequence must be
    Active -> Recycling -> Active.
    """
    clock = FakeClock()
    target = make_target_info(filename_hint="file.bin", content_length=1_000_000)
    server = FakeAria2Server(
        gid="gid-e2e",
        status="active",
        completed_length=0,
        total_length=target.content_length,
        download_speed=200_000.0,
        servers=[{"servers": [{"downloadSpeed": "200000"}]}],
    )
    watchdog = WatchdogController(
        mode=WatchdogMode.V2,
        user_floor_bps=100_000.0,
        grace_period=2,
        connections=16,
        supports_range=True,
        min_cooldown_sec=2.0,
        recycle_interval_sec=5.0,
        hour_bucket=0,
        network_id="e2e-test-net",
        apply_learned_profile=False,
    )

    emitted: list[tuple[str, dict[str, Any]]] = []

    def _capture(event_type: str, **payload: Any) -> None:
        emitted.append((event_type, payload))

    monkeypatch.setattr("paradl.ui.UI.emit_event", _capture)

    states: list[str] = []
    observed_statuses: list[str] = []
    paused_snapshot: dict[str, Any] = {}
    t0 = clock.monotonic()

    def _tick(elapsed_sec: float) -> WatchdogDecision:
        nonlocal paused_snapshot
        # Simulate multi-second elapsed time via the fake clock: ZERO real sleeps.
        clock.advance(elapsed_sec - (clock.monotonic() - t0))
        # Per-tick progress/status supplied by the fake aria2 daemon.
        stat = server.get_global_stat()
        status = server.tell_status(server.gid)
        observed_statuses.append(str(status["status"]))
        per_conn = [float(s["downloadSpeed"]) for s in server.get_servers()[0]["servers"]]
        decision = watchdog.step(
            elapsed_sec=clock.monotonic() - t0,
            speed_bps=float(stat["downloadSpeed"]),
            per_conn_speeds=per_conn,
        )
        if decision.action != WatchdogActionType.NONE:
            states.append("Recycling")
            # Emit the recycle event through the real UI surface (mirrors aria2.py).
            UI.emit_event(
                "watchdog_action",
                cycle=decision.cycle,
                mode=watchdog.mode.value,
                action=decision.action.value,
                connections=watchdog.connections,
            )
            # Execute the recycle against the fake daemon.
            server.force_pause(server.gid)
            server.update(status="paused")
            paused_snapshot = server.tell_status(server.gid)
            clock.sleep(0.15)  # fake pause/unpause gap -- no real sleep
            server.unpause(server.gid)
            server.update(status="active")
        else:
            states.append("Active" if watchdog.is_active else "Inactive")
        return decision

    # Active: healthy traffic below the periodic recycle interval.
    assert watchdog.is_active
    assert _tick(0.0).action == WatchdogActionType.NONE

    # Recycling: after >5 s of fake elapsed time the periodic socket cycle fires.
    recycle = _tick(5.5)
    assert recycle.action == WatchdogActionType.PAUSE_UNPAUSE
    assert recycle.cycle == 1

    # Active again: cooldown elapsed, monitoring resumes without another cycle.
    assert _tick(8.0).action == WatchdogActionType.NONE
    assert watchdog.is_active

    # End-to-end composition assertions (no single unit-assertion duplicate).
    assert states == ["Active", "Recycling", "Active"]
    watchdog_events = [payload for name, payload in emitted if name == "watchdog_action"]
    assert len(watchdog_events) == 1
    assert watchdog_events[0]["action"] == WatchdogActionType.PAUSE_UNPAUSE.value

    # FakeAria2Server was actually exercised for progress/status and controls.
    assert "get_global_stat" in server.calls
    assert "tell_status" in server.calls
    assert "get_servers" in server.calls
    assert server.calls.count("force_pause") == 1
    assert server.calls.count("unpause") == 1
    assert paused_snapshot["status"] == "paused"
    assert observed_statuses == ["active", "active", "active"]

    # Multi-second time was simulated entirely with the fake clock.
    assert clock.monotonic() - t0 == pytest.approx(8.0)
    assert 0.15 in clock.sleep_history

class _CountingDaemon:
    """Fake Aria2RpcDaemon that reports `active` for two ticks, then `complete`."""

    def __init__(self, gid: str, out_file: Path) -> None:
        self.gid = gid
        self.out_file = out_file
        self.tell_calls = 0
        self.server_calls = 0
        self.global_stat_calls = 0
        self.current_tick = 0

    def add_uri(self, uris: list[str], options: dict) -> str:
        return self.gid

    def get_global_stat(self) -> dict:
        # Called exactly once per poll tick -- use it to advance the fake clock.
        self.global_stat_calls += 1
        self.current_tick = self.global_stat_calls
        return {"downloadSpeed": "1024"}

    def tell_status(self, gid: str) -> dict:
        self.tell_calls += 1
        if self.current_tick >= 3:
            return {
                "status": "complete",
                "completedLength": "1000",
                "totalLength": "1000",
                "files": [{"path": str(self.out_file)}],
            }
        return {
            "status": "active",
            "completedLength": "500",
            "totalLength": "1000",
            "files": [
                {
                    "path": str(self.out_file),
                    "completedLength": "500",
                    "length": "1000",
                }
            ],
        }

    def get_servers(self, gid: str) -> list[dict]:
        self.server_calls += 1
        return [{"servers": [{"downloadSpeed": "500"}]}]

    def stop(self, *, graceful: bool = True) -> None:
        return None

"""Wave-2 concurrency parity tests.

These tests pin three invariants before/after the bounded-concurrency changes:

1. API task dispatch never runs more workers than ``max_concurrent`` (atomic
   slot reservation), even when many dispatches happen at once.
2. Multi-file Hugging Face downloads preserve input order and still call the
   auth-carrying downloader and SHA-256 verifier exactly once per file.
3. PyPI dependency pre-fetch preserves input order and still runs the
   SHA-256-verifying ``execute()`` path exactly once per wheel.

All fakes are local; no network, real downloads, or real subprocesses occur.
"""

from __future__ import annotations

import sys
import threading
import types
from pathlib import Path
from typing import Any

import pytest

from paradl.core.models import TargetType, TransferResult

# ---------------------------------------------------------------------------
# 1. API dispatch slot safety (deterministic, barrier-based)
# ---------------------------------------------------------------------------


class _FakeStore:
    """Minimal _TaskStore surface used by _TaskRunner._spawn_worker()."""

    def __init__(self) -> None:
        self.threads: dict[str, threading.Thread] = {}
        self.stop_events: dict[str, threading.Event] = {}
        self.updates: list[tuple[str, dict[str, Any]]] = []

    def set_thread(self, task_id: str, thread: threading.Thread) -> None:
        self.threads[task_id] = thread

    def set_stop_event(self, task_id: str, event: threading.Event) -> None:
        self.stop_events[task_id] = event

    def update(self, task_id: str, **kwargs: Any) -> None:
        self.updates.append((task_id, kwargs))


class TestApiDispatchSlotSafety:
    def _make_runner(self, max_concurrent: int = 3):
        from paradl.api.server import _TaskRunner

        store = _FakeStore()
        runner = _TaskRunner(store=store, config=object(), max_concurrent=max_concurrent)  # type: ignore[arg-type]
        return runner, store

    def test_concurrent_dispatch_never_exceeds_max(self) -> None:
        """N simultaneous dispatch attempts must reserve at most max slots."""
        runner, _store = self._make_runner(max_concurrent=3)

        release = threading.Event()
        peak = {"value": 0}
        count_lock = threading.Lock()
        spawned: list[threading.Thread | None] = []
        barrier = threading.Barrier(16)

        # Replace the real _execute with a fake that records the active count
        # while it "runs", then releases its slot in a finally block (matching
        # the real _execute contract).
        def fake_execute(task: Any) -> None:
            with count_lock:
                peak["value"] = max(peak["value"], runner._active_count())
            try:
                release.wait(timeout=5.0)
            finally:
                runner._dec_active()

        runner._execute = fake_execute  # type: ignore[method-assign]

        def dispatch(idx: int) -> None:
            task = types.SimpleNamespace(id=f"task-{idx:02d}")
            barrier.wait()  # all dispatch attempts start together
            spawned.append(runner._spawn_worker(task))

        threads = [threading.Thread(target=dispatch, args=(i,)) for i in range(16)]
        for t in threads:
            t.start()
        for t in threads:
            t.join(timeout=10.0)

        # Exactly max workers were spawned; the rest saw a full pool.
        assert sum(1 for w in spawned if w is not None) == 3
        assert peak["value"] <= 3
        assert runner._active_count() == 3

        # Let the workers finish; every reserved slot is returned.
        release.set()
        for w in spawned:
            if w is not None:
                w.join(timeout=10.0)
        assert runner._active_count() == 0

    def test_released_slot_can_be_reused(self) -> None:
        """After a worker finishes, a new dispatch must be able to reserve."""
        runner, _store = self._make_runner(max_concurrent=1)

        release = threading.Event()

        def fake_execute(task: Any) -> None:
            try:
                release.wait(timeout=5.0)
            finally:
                runner._dec_active()

        runner._execute = fake_execute  # type: ignore[method-assign]

        worker_a = runner._spawn_worker(types.SimpleNamespace(id="task-a"))
        assert worker_a is not None
        assert runner._spawn_worker(types.SimpleNamespace(id="task-b")) is None

        release.set()
        worker_a.join(timeout=5.0)
        assert runner._active_count() == 0

        worker_c = runner._spawn_worker(types.SimpleNamespace(id="task-c"))
        assert worker_c is not None
        worker_c.join(timeout=5.0)

    def test_spawn_failure_releases_reserved_slot(self, monkeypatch: pytest.MonkeyPatch) -> None:
        """If worker startup fails, the reserved slot must be returned.

        A RuntimeError from Thread.start() (e.g. thread exhaustion) means
        _execute never runs, so _spawn_worker itself must release the slot;
        otherwise the active count stays high and _wait_for_slot() can block
        forever.
        """
        from paradl.api import server as server_mod

        runner, _store = self._make_runner(max_concurrent=1)
        runner._execute = lambda task: runner._dec_active()  # type: ignore[method-assign]

        def boom(self: threading.Thread) -> None:
            raise RuntimeError("can't start new thread")

        with monkeypatch.context() as m:
            m.setattr(server_mod.threading.Thread, "start", boom)
            with pytest.raises(RuntimeError, match="can't start new thread"):
                runner._spawn_worker(types.SimpleNamespace(id="task-fail"))
            # The slot reserved before the failure was returned.
            assert runner._active_count() == 0

        # With start() restored, a later dispatch is not blocked by a leak.
        worker = runner._spawn_worker(types.SimpleNamespace(id="task-ok"))
        assert worker is not None
        worker.join(timeout=5.0)
        assert runner._active_count() == 0

class TestPypiConcurrentParity:
    def test_batch_results_preserve_order_and_verify_each(
        self, tmp_path: Path, monkeypatch: pytest.MonkeyPatch
    ) -> None:
        from paradl.engines import aria2 as aria2_mod
        from paradl.engines import pypi as pypi_mod

        deps = ["dep-a==1.0", "dep-b==2.0", "dep-c", "dep-d==4.0"]

        def fake_probe(target: str, py_version: Any = None) -> Any:
            name = target.split("pip:", 1)[-1].split("==", 1)[0]
            return types.SimpleNamespace(
                target_type=TargetType.PYPI,
                clean_url=f"https://example.invalid/{name}.whl",
                pypi_meta={"package": name, "version": "1.0", "sha256": "ab" * 32},
            )

        # Real PyPIEngine.execute() runs, but its Aria2 download is faked so it
        # produces a real on-disk file. The real execute() then performs the
        # SHA-256 verification branch, which we count below.
        def fake_aria2_download(*args: Any, **kwargs: Any) -> TransferResult:
            out_dir = kwargs.get("out_dir") or (args[1] if len(args) > 1 else tmp_path)
            path = Path(out_dir) / "fake.whl"
            if not path.exists():

def _dump_schema(db_file: Path) -> dict[str, object]:
    """Capture the complete on-disk schema plus connection-level pragmas."""
    conn = sqlite3.connect(str(db_file))
    conn.row_factory = sqlite3.Row
    try:
        master = [
            (row["type"], row["name"], row["sql"])
            for row in conn.execute(
                "SELECT type, name, sql FROM sqlite_master ORDER BY type, name"
            ).fetchall()
        ]
        tables = [
            row["name"]
            for row in conn.execute(
                "SELECT name FROM sqlite_master WHERE type = 'table' ORDER BY name"
            ).fetchall()
        ]
        table_info = {
            table: [
                (
                    row["cid"],
                    row["name"],
                    row["type"],
                    row["notnull"],
                    row["dflt_value"],
                    row["pk"],
                )
                for row in conn.execute(f"PRAGMA table_info({table})").fetchall()
            ]
            for table in tables
        }
        pragmas = {
            "journal_mode": conn.execute("PRAGMA journal_mode").fetchone()[0],
            "synchronous": conn.execute("PRAGMA synchronous").fetchone()[0],
            "user_version": conn.execute("PRAGMA user_version").fetchone()[0],
        }
        return {"master": master, "table_info": table_info, "pragmas": pragmas}
    finally:
        conn.close()


def _table_names(db_file: Path) -> set[str]:
    conn = sqlite3.connect(str(db_file))
    try:
        return {
            row[0]
            for row in conn.execute("SELECT name FROM sqlite_master WHERE type = 'table'")
        }
    finally:
        conn.close()


def _dump_profile_rows(db_file: Path) -> tuple[list[tuple], list[tuple]]:
    conn = sqlite3.connect(str(db_file))
    try:

        strategies = conn.execute(
            "SELECT * FROM host_strategies ORDER BY host"
        ).fetchall()
        return [tuple(row) for row in learned], [tuple(row) for row in strategies]
    finally:
        conn.close()

def test_schema_unchanged_across_batch_of_connections():
    """Schema, pragmas and schema_meta must not change after a batch of fresh connections."""
    db_file = get_history_db_file()
    with contextlib.closing(HistoryManager._connect()):
        pass
    with ThrottleStore._conn():
        pass
    before = _dump_schema(db_file)

    for index in range(5):
        HistoryManager.record(f"https://example.com/{index}", "direct", None, None, 1.0, True)
        ThrottleStore.set_cached_asn(f"9.9.9.{index}", "AS9", "US", "Example")

    after = _dump_schema(db_file)
    assert after == before
    assert after["pragmas"]["journal_mode"] == "wal"

    # WAL/synchronous pragmas are per-connection: a managed connection still applies them.

        assert managed.execute("PRAGMA journal_mode").fetchone()[0] == "wal"
        assert managed.execute("PRAGMA synchronous").fetchone()[0] == 1

    conn = sqlite3.connect(str(db_file))
    try:
        version = conn.execute(
            "SELECT value FROM schema_meta WHERE key = 'schema_version'"
        ).fetchone()
    finally:
        conn.close()
    assert version is not None
    assert version[0] == str(HISTORY_SCHEMA_VERSION)


def test_ddl_executed_at_most_once_for_repeated_connections(monkeypatch: pytest.MonkeyPatch):
    """Repeated connections to one DB file must not re-run the CREATE TABLE DDL."""
    statements: list[str] = []
    real_connect = sqlite3.connect

    def traced_connect(*args, **kwargs):  # type: ignore[no-untyped-def]
        conn = real_connect(*args, **kwargs)
        conn.set_trace_callback(statements.append)
        return conn

    with monkeypatch.context() as patcher:
        patcher.setattr("paradl.storage.history.sqlite3.connect", traced_connect)
        for _ in range(5):
            with contextlib.closing(HistoryManager._connect()):
                pass
            with ThrottleStore._conn():
                pass

    create_statements = [
        statement
        for statement in statements
        if statement.strip().upper().startswith("CREATE")
    ]
    # 7 history tables + 3 throttle tables, initialized exactly once per process/file.
    assert len(create_statements) <= 10, create_statements


def test_reopen_after_delete_recreates_schema():
    """Deleting/recreating the DB file invalidates the once-per-process guard."""
    db_file = get_history_db_file()
    with contextlib.closing(HistoryManager._connect()):
        pass
    with ThrottleStore._conn():
        pass
    before = _dump_schema(db_file)

    db_file.unlink()
    for suffix in ("-wal", "-shm"):
        side = Path(str(db_file) + suffix)
        if side.exists():
            side.unlink()
    assert not db_file.exists()

    with contextlib.closing(HistoryManager._connect()) as conn:
        names = {row[0] for row in conn.execute("SELECT name FROM sqlite_master WHERE type='table'")}
    assert names >= HISTORY_TABLES

    with ThrottleStore._conn() as conn:
        names = {row[0] for row in conn.execute("SELECT name FROM sqlite_master WHERE type='table'")}
    assert names >= THROTTLE_TABLES

    assert _dump_schema(db_file) == before

def test_batched_persistence_matches_one_by_one(
    monkeypatch: pytest.MonkeyPatch, tmp_path: Path
):
    """Rows written by the batched path must be byte-for-byte identical to per-profile saves."""
    monkeypatch.setattr(
        "paradl.throttle.store.get_clock", lambda: FakeClock(FIXED_NOW)
    )
    profile, targets, dataset = _fixture_inputs()
    expected = _expected_profiles(profile, targets, dataset)

    # Baseline: save each profile independently (legacy behaviour).
    monkeypatch.setenv("PARADL_HOME", str(tmp_path / "one_by_one"))
    db_one = get_history_db_file()
    with contextlib.closing(HistoryManager._connect()):
        pass
    for learned in expected:
        HistoryManager.save_learned_profile(learned)
    baseline = _dump_profile_rows(db_one)

    # Batched path: one call writes the same rows.
    monkeypatch.setenv("PARADL_HOME", str(tmp_path / "batched"))
    db_batch = get_history_db_file()
    with contextlib.closing(HistoryManager._connect()):
        pass
    ThrottleStore.persist_recommended_profile(profile, targets, dataset)
    batched = _dump_profile_rows(db_batch)

    assert batched == baseline
    assert len(batched[0]) == len(expected)
